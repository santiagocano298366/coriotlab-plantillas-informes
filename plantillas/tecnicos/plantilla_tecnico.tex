%% TÍTULO: Nombre completo y descriptivo del informe técnico.
%%   Máximo 100 caracteres. Debe identificar claramente el sistema o
%%   proyecto documentado. Aparece en portada y encabezado.
%%   Ejemplo: "Sistema de Monitoreo Ambiental Distribuido para CORIOTLAB"
\newcommand{\DocTitulo}{[Título completo del informe técnico]}

%% SUBTÍTULO: Complemento técnico del título. Describe el enfoque
%%   metodológico, tecnológico o el alcance del documento.
%%   Ejemplo: "Diseño, implementación y validación de una red IoT con ESP32"
\newcommand{\DocSubtitulo}{[Subtítulo técnico: enfoque o alcance del proyecto]}

%% NÚMERO DE DOCUMENTO: Identificador único del informe.
%%   Formato: IT-AAAA-NNN
%%   IT = Informe Técnico | AAAA = año | NNN = correlativo (001, 002…)
%%   Ejemplo: IT-2025-001
\newcommand{\DocNumero}{IT-2025-[NNN]}

%% PROYECTO: Nombre oficial del proyecto al que pertenece el informe.
%%   Debe coincidir con el registro en el portafolio de CORIOTLAB.
\newcommand{\DocProyecto}{[Nombre oficial del proyecto]}

%% CÓDIGO INTERNO: Código de seguimiento del proyecto.
%%   Formato sugerido: CORIOT-AAAA-SIGLAS-NNN
%%   Ejemplo: CORIOT-2025-MA-001
\newcommand{\DocCodigo}{CORIOT-2025-[SIGLAS]-[NNN]}

%% CLIENTE / ENTIDAD: Organización o persona que recibe el informe.
%%   Ejemplo: "Laboratorio CORIOTLAB --- Facultad de Ingeniería, ITM"
\newcommand{\DocCliente}{[Nombre del cliente o entidad destinataria]}

%% PERÍODO: Rango de fechas que cubre el proyecto o fase documentada.
%%   Formato: "Mes --- Mes AAAA" o "DD de mes --- DD de mes AAAA"
%%   Ejemplo: "Enero --- Junio 2025"
\newcommand{\DocPeriodo}{[Mes inicial] --- [Mes final] [AAAA]}

%% LUGAR: Ciudad y país donde se desarrolló el proyecto.
%%   Ejemplo: "Medellín, Colombia"
\newcommand{\DocLugar}{Medellín, Colombia}

%% VERSIÓN: Versión del documento. Iniciar en 1.0.
%%   Incrementar decimalmente en revisiones; entero en reestructuraciones.
\newcommand{\DocVersion}{1.0}

%% FECHA: Fecha de emisión del informe.
%%   Formato: DD de mes de AAAA — Ejemplo: 30 de junio de 2025
\newcommand{\DocFecha}{[DD de mes de AAAA]}

%% CONFIDENCIALIDAD: Nivel de acceso del documento.
%%   Valores: "Uso interno --- CORIOTLAB" | "Confidencial" |
%%            "Uso restringido" | "Público"
\newcommand{\DocConfidencialidad}{Uso interno --- CORIOTLAB}

%% AUTOR: Nombre completo del autor principal.
\newcommand{\DocAutor}{[Nombre completo del autor]}

%% CARGO DEL AUTOR: Rol del autor en el proyecto / CORIOTLAB.
%%   Ejemplo: "Investigador Principal --- CORIOTLAB"
\newcommand{\DocCargoAutor}{[Cargo --- CORIOTLAB]}

%% REVISOR: Nombre completo de quien revisa y aprueba el informe.
%%   Generalmente el director del laboratorio o líder del proyecto.
\newcommand{\DocRevisor}{[Nombre del revisor]}

%% CARGO DEL REVISOR: Rol del revisor.
%%   Ejemplo: "Director de Laboratorio --- CORIOTLAB"
\newcommand{\DocCargoRevisor}{[Cargo del revisor --- CORIOTLAB]}

%% ENTIDAD: Institución a la que pertenecen autor y revisor.
%%   Ejemplo: "CORIOTLAB --- ITM"
\newcommand{\DocEntidad}{CORIOTLAB --- ITM}
