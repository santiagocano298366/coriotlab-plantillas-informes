%% VARIANTE ACTIVA: Controla qué sección del documento se compila.
%%   "periodico" → activa tabla de actividades con estados.
%%   "guia"      → activa documentación de procedimiento técnico.
%% Para cambiar de variante: 1) edite este valor, 2) comente la sección
%% activa del cuerpo y descomente la sección correspondiente.
\newcommand{\TipoVariante}{periodico}

%% TIPO DE PERÍODO (solo para Variante Periódico):
%%   Valores permitidos: Semanal | Quincenal | Mensual
%%   Aparece en el encabezado de la portada: "Informe Semanal", etc.
\newcommand{\TipoPeriodo}{Semanal}

%% TÍTULO: Nombre descriptivo del proyecto o trabajo documentado.
%%   Máximo 80 caracteres. Evitar abreviaciones.
%%   Aparece en portada y en el encabezado de cada página.
%%   Ejemplo: "Sistema de Monitoreo Ambiental con ESP32"
\newcommand{\DocTitulo}{[Título del informe o proyecto]}

%% NÚMERO DE DOCUMENTO: Identificador único.
%%   Formato: IA-AAAA-NNN
%%   IA = Informe de Actividades | AAAA = año | NNN = número correlativo (001, 002…)
%%   Ejemplo: IA-2025-001
\newcommand{\DocNumero}{IA-2025-[NNN]}

%% AUTOR: Nombre completo de quien elabora el informe.
%%   Aparece en portada y en la firma al final.
\newcommand{\DocAutor}{[Nombre completo del autor]}

%% CARGO: Rol del autor dentro de CORIOTLAB.
%%   Valores frecuentes: Investigador | Investigador Principal |
%%                       Monitor | Director | Auxiliar de laboratorio
\newcommand{\DocCargo}{[Cargo --- CORIOTLAB]}

%% PROYECTO: Nombre oficial del proyecto al que pertenece este informe.
%%   Debe coincidir con el nombre registrado en el plan de trabajo.
\newcommand{\DocProyecto}{[Nombre del proyecto]}

%% PERÍODO REPORTADO: Rango de fechas del período documentado.
%%   Formato libre pero coherente. Ejemplo: "16 al 20 de junio de 2025"
\newcommand{\DocPeriodo}{[Período reportado: ej. "16 al 20 de junio de 2025"]}

%% FECHA DE ENTREGA: Fecha en que se entrega el informe.
%%   Formato: DD de mes de AAAA — Ejemplo: 20 de junio de 2025
%%   Use \today para fecha automática del sistema, o escriba la fecha fija.
\newcommand{\DocFecha}{[DD de mes de AAAA]}

%% VERSIÓN: Versión del documento. Iniciar en 1.0.
%%   Incrementar decimalmente en revisiones menores (1.1, 1.2…);
%%   incrementar entero en reestructuraciones mayores (2.0).
\newcommand{\DocVersion}{1.0}

\newcommand{\PortadaActividades}[2]{%
  \thispagestyle{empty}
  \begin{tikzpicture}[remember picture, overlay]
    \fill[AzulITM] (current page.north west)
      rectangle ([yshift=-4.5cm] current page.north east);
    \node[anchor=west, text=white] at
      ([xshift=2.5cm, yshift=-1.5cm] current page.north west)
      {\FuenteTitulo\large\textbf{#1}};
    \node[anchor=east, text=AzulDigital] at
      ([xshift=-2.5cm, yshift=-1.5cm] current page.north east)
      {\FuenteTitulo\Large\textbf{#2}};
    \draw[AzulDigital, line width=2pt]
      ([yshift=-4.5cm] current page.north west) --
      ([yshift=-4.5cm] current page.north east);
  \end{tikzpicture}%
  \vspace{3.8cm}

  \begin{center}
    {\FuenteTitulo\Huge\color{AzulITM}\textbf{\DocTitulo}}\\[0.6em]
    {\FuenteTitulo\large\color{GrisPizarra}\DocProyecto}\\[2em]
    {\color{GrisLinea}\rule{\linewidth}{0.6pt}}\\[1.2em]
